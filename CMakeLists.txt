cmake_minimum_required(VERSION 3.13)
project(minifetch LANGUAGES CXX)

option(MINIFETCH_BUILD_TEST "Build the example application" ON)
option(MINIFETCH_BUILD_REGRESSION_TESTS "Build local regression tests" OFF)

add_library(minifetch STATIC src/minifetch.cpp src/minifetch.h)
target_compile_features(minifetch PUBLIC cxx_std_17)
target_include_directories(minifetch PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
if(WIN32)
    target_link_libraries(minifetch PRIVATE winhttp)
elseif(EMSCRIPTEN)
    target_compile_options(minifetch PUBLIC "-fexceptions")
    target_link_options(minifetch PUBLIC "-sFETCH=1" "-sASYNCIFY=1" "-fexceptions")
else()
    find_package(CURL REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(minifetch PRIVATE CURL::libcurl Threads::Threads)
endif()

if(MINIFETCH_BUILD_TEST)
    add_executable(minifetchtest src/main.cpp)
    target_compile_definitions(minifetchtest PRIVATE MINIFETCH_BUILD_TEST)
    target_link_libraries(minifetchtest PRIVATE minifetch)
endif()

if(MINIFETCH_BUILD_REGRESSION_TESTS AND NOT EMSCRIPTEN)
    enable_testing()
    find_package(Threads REQUIRED)
    if(NOT WIN32)
        add_executable(minifetch_browser_contract tests/browser_backend.cpp src/minifetch.cpp)
        target_compile_features(minifetch_browser_contract PRIVATE cxx_std_17)
        target_compile_definitions(minifetch_browser_contract PRIVATE __EMSCRIPTEN__)
        target_include_directories(minifetch_browser_contract PRIVATE tests/emscripten_stubs src)
        target_link_libraries(minifetch_browser_contract PRIVATE Threads::Threads)
        add_test(NAME minifetch_browser_contract COMMAND minifetch_browser_contract)
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(minifetch_regression tests/regression.cpp)
    target_link_libraries(minifetch_regression PRIVATE minifetch Threads::Threads)
    add_test(NAME minifetch_regression COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/tests/server.py" "$<TARGET_FILE:minifetch_regression>")
    set_tests_properties(minifetch_regression PROPERTIES TIMEOUT 45)
endif()
